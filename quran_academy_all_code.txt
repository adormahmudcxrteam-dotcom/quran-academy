name: Build Quran Academy APK

on:
  push:
  workflow_dispatch:

jobs:
  build-apk:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up JDK 21
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'

      - name: Accept Android SDK licenses
        run: |
          if [ -x "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" ]; then
            yes | "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" --licenses > /dev/null || true
          fi

      - name: Ensure debug keystore exists
        run: |
          if [ ! -f debug.keystore ]; then
            keytool -genkeypair -v \
              -keystore debug.keystore \
              -storepass android \
              -alias androiddebugkey \
              -keypass android \
              -keyalg RSA \
              -keysize 2048 \
              -validity 10000 \
              -dname "CN=Android Debug,O=Android,C=US"
          fi

      - name: Set up Gradle 9.3.1
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: 9.3.1

      - name: Build Debug APK
        run: gradle :app:assembleDebug --no-daemon

      - name: Upload Quran Academy APK
        uses: actions/upload-artifact@v4
        with:
          name: Quran-Academy-APK
          path: app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: error

# Run and deploy your AI Studio app

This contains everything you need to run your app locally.

View your app in AI Studio: https://ai.studio/apps/a3119f2a-916c-4eeb-8293-ffbce19df953

## Run Locally

**Prerequisites:**  [Android Studio](https://developer.android.com/studio)


1. Open Android Studio
2. Select **Open** and choose the directory containing this project
3. Allow Android Studio to fix any incompatibilities as it imports the project.
4. Create a file named `.env` in the project directory and set `GEMINI_API_KEY` in that file to your Gemini API key (see `.env.example` for an example)
5. Remove this line from the app's `build.gradle.kts` file: `signingConfig = signingConfigs.getByName("debugConfig")`
6. Run the app on an emulator or physical device
7. If you have already published your app in AI Studio, please [request upload key reset](https://support.google.com/googleplay/android-developer/answer/9842756#zippy=%2Crequest-an-upload-key-reset) in Google Play Console.

package com.example

import android.graphics.Color
import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.SystemBarStyle
import androidx.activity.compose.setContent
import androidx.activity.enableEdgeToEdge
import androidx.activity.viewModels
import androidx.compose.runtime.SideEffect
import androidx.compose.runtime.getValue
import androidx.lifecycle.compose.collectAsStateWithLifecycle
import com.example.ui.navigation.AcademyApp
import com.example.ui.theme.QuranAcademyTheme
import com.example.viewmodel.AcademyViewModel

class MainActivity : ComponentActivity() {
    private val academyViewModel: AcademyViewModel by viewModels()

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContent {
            val userProgress by academyViewModel.userProgress.collectAsStateWithLifecycle()
            val darkTheme = userProgress.isDarkMode

            SideEffect {
                enableEdgeToEdge(
                    statusBarStyle = if (darkTheme) {
                        SystemBarStyle.dark(Color.TRANSPARENT)
                    } else {
                        SystemBarStyle.light(Color.TRANSPARENT, Color.TRANSPARENT)
                    },
                    navigationBarStyle = if (darkTheme) {
                        SystemBarStyle.dark(Color.TRANSPARENT)
                    } else {
                        SystemBarStyle.light(Color.TRANSPARENT, Color.TRANSPARENT)
                    }
                )
            }

            QuranAcademyTheme(darkTheme = darkTheme) {
                AcademyApp(viewModel = academyViewModel)
            }
        }
    }
}

package com.example.data

import android.content.Context
import androidx.datastore.core.DataStore
import androidx.datastore.preferences.core.Preferences
import androidx.datastore.preferences.core.booleanPreferencesKey
import androidx.datastore.preferences.core.edit
import androidx.datastore.preferences.core.floatPreferencesKey
import androidx.datastore.preferences.core.intPreferencesKey
import androidx.datastore.preferences.core.stringSetPreferencesKey
import androidx.datastore.preferences.preferencesDataStore
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.map

private val Context.academyDataStore: DataStore<Preferences> by preferencesDataStore(
    name = "quran_learning_academy_prefs"
)

class AcademyPreferencesRepository(private val context: Context) {

    private object Keys {
        val LAST_READ_SURAH = intPreferencesKey("last_read_surah")
        val BOOKMARKED_AYAHS = stringSetPreferencesKey("bookmarked_ayahs")
        val BOOKMARKED_HADITHS = stringSetPreferencesKey("bookmarked_hadiths")
        val BOOKMARKED_DUAS = stringSetPreferencesKey("bookmarked_duas")
        val STUDIED_SURAHS = stringSetPreferencesKey("studied_surahs")
        val COMPLETED_LESSONS = stringSetPreferencesKey("completed_lessons")
        val COUNTER_VALUE = intPreferencesKey("tasbih_count")
        val COUNTER_TARGET = intPreferencesKey("tasbih_target")
        val TOTAL_REMEMBRANCE = intPreferencesKey("total_dhikr")
        val BEST_QUIZ_SCORE = intPreferencesKey("best_quiz_score")
        val READING_FONT_SIZE = floatPreferencesKey("reading_font_size")
        val SHOW_STUDY_NOTES = booleanPreferencesKey("show_study_notes")
        val DARK_MODE = booleanPreferencesKey("dark_mode")
    }

    val userProgressFlow: Flow<UserProgress> = context.academyDataStore.data.map { prefs ->
        UserProgress(
            lastReadSurahNumber = prefs[Keys.LAST_READ_SURAH] ?: 1,
            bookmarkedAyahKeys = prefs[Keys.BOOKMARKED_AYAHS] ?: emptySet(),
            bookmarkedHadithIds = prefs[Keys.BOOKMARKED_HADITHS] ?: emptySet(),
            bookmarkedDuaIds = prefs[Keys.BOOKMARKED_DUAS] ?: emptySet(),
            studiedSurahNumbers = (prefs[Keys.STUDIED_SURAHS] ?: emptySet())
                .mapNotNull { it.toIntOrNull() }
                .toSet(),
            completedLessonIds = prefs[Keys.COMPLETED_LESSONS] ?: emptySet(),
            counterValue = prefs[Keys.COUNTER_VALUE] ?: 0,
            counterTarget = prefs[Keys.COUNTER_TARGET] ?: 33,
            totalRemembranceCount = prefs[Keys.TOTAL_REMEMBRANCE] ?: 0,
            bestQuizScore = prefs[Keys.BEST_QUIZ_SCORE] ?: 0,
            readingFontSizeSp = prefs[Keys.READING_FONT_SIZE] ?: 18f,
            showStudyNotes = prefs[Keys.SHOW_STUDY_NOTES] ?: true,
            isDarkMode = prefs[Keys.DARK_MODE] ?: false
        )
    }

    suspend fun setLastReadSurah(surahNumber: Int) {
        context.academyDataStore.edit { prefs ->
            prefs[Keys.LAST_READ_SURAH] = surahNumber
        }
    }

    suspend fun toggleAyahBookmark(ayahKey: String) {
        context.academyDataStore.edit { prefs ->
            val current = prefs[Keys.BOOKMARKED_AYAHS] ?: emptySet()
            prefs[Keys.BOOKMARKED_AYAHS] = if (ayahKey in current) current - ayahKey else current + ayahKey
        }
    }

    suspend fun toggleHadithBookmark(hadithId: String) {
        context.academyDataStore.edit { prefs ->
            val current = prefs[Keys.BOOKMARKED_HADITHS] ?: emptySet()
            prefs[Keys.BOOKMARKED_HADITHS] = if (hadithId in current) current - hadithId else current + hadithId
        }
    }

    suspend fun toggleDuaBookmark(duaId: String) {
        context.academyDataStore.edit { prefs ->
            val current = prefs[Keys.BOOKMARKED_DUAS] ?: emptySet()
            prefs[Keys.BOOKMARKED_DUAS] = if (duaId in current) current - duaId else current + duaId
        }
    }

    suspend fun toggleSurahStudied(surahNumber: Int) {
        context.academyDataStore.edit { prefs ->
            val key = surahNumber.toString()
            val current = prefs[Keys.STUDIED_SURAHS] ?: emptySet()
            prefs[Keys.STUDIED_SURAHS] = if (key in current) current - key else current + key
        }
    }

    suspend fun toggleLessonCompleted(lessonId: String) {
        context.academyDataStore.edit { prefs ->
            val current = prefs[Keys.COMPLETED_LESSONS] ?: emptySet()
            prefs[Keys.COMPLETED_LESSONS] = if (lessonId in current) current - lessonId else current + lessonId
        }
    }

    suspend fun incrementCounter() {
        context.academyDataStore.edit { prefs ->
            val target = prefs[Keys.COUNTER_TARGET] ?: 33
            val current = prefs[Keys.COUNTER_VALUE] ?: 0
            val next = if (current + 1 > target) 1 else current + 1
            prefs[Keys.COUNTER_VALUE] = next
            prefs[Keys.TOTAL_REMEMBRANCE] = (prefs[Keys.TOTAL_REMEMBRANCE] ?: 0) + 1
        }
    }

    suspend fun resetCounter() {
        context.academyDataStore.edit { prefs ->
            prefs[Keys.COUNTER_VALUE] = 0
        }
    }

    suspend fun setCounterTarget(target: Int) {
        context.academyDataStore.edit { prefs ->
            prefs[Keys.COUNTER_TARGET] = target
            prefs[Keys.COUNTER_VALUE] = 0
        }
    }

    suspend fun recordQuizScore(score: Int) {
        context.academyDataStore.edit { prefs ->
            val best = prefs[Keys.BEST_QUIZ_SCORE] ?: 0
            if (score > best) {
                prefs[Keys.BEST_QUIZ_SCORE] = score
            }
        }
    }

    suspend fun setReadingFontSize(sizeSp: Float) {
        context.academyDataStore.edit { prefs ->
            prefs[Keys.READING_FONT_SIZE] = sizeSp.coerceIn(15f, 26f)
        }
    }

    suspend fun setShowStudyNotes(show: Boolean) {
        context.academyDataStore.edit { prefs ->
            prefs[Keys.SHOW_STUDY_NOTES] = show
        }
    }

    suspend fun toggleDarkMode() {
        context.academyDataStore.edit { prefs ->
            val current = prefs[Keys.DARK_MODE] ?: false
            prefs[Keys.DARK_MODE] = !current
        }
    }
}

 */
@Composable
fun RubElHizbBadge(
    numberText: String,
    modifier: Modifier = Modifier,
    size: Dp = 42.dp,
    strokeColor: Color = MaterialTheme.colorScheme.primary,
    fillColor: Color = MaterialTheme.colorScheme.primaryContainer.copy(alpha = 0.55f),
    textColor: Color = MaterialTheme.colorScheme.primary
) {
    Box(
        modifier = modifier.size(size),
        contentAlignment = Alignment.Center
    ) {
        Canvas(modifier = Modifier.fillMaxSize()) {
            val side = this.size.minDimension * 0.68f
            val topLeft = Offset((this.size.width - side) / 2f, (this.size.height - side) / 2f)
            val squareSize = Size(side, side)
            val stroke = Stroke(width = 1.6.dp.toPx())

            drawRect(color = fillColor, topLeft = topLeft, size = squareSize)
            drawRect(color = strokeColor, topLeft = topLeft, size = squareSize, style = stroke)

            rotate(degrees = 45f) {
                drawRect(color = fillColor, topLeft = topLeft, size = squareSize)
                drawRect(color = strokeColor, topLeft = topLeft, size = squareSize, style = stroke)
            }
        }
        Text(
            text = numberText,
            style = MaterialTheme.typography.labelMedium,
            fontWeight = FontWeight.Bold,
            color = textColor,
            maxLines = 1
        )
    }
}

/**
 * Refined horizontal ornamental divider with a central diamond motif.
 */
@Composable
fun OrnamentalDivider(
    modifier: Modifier = Modifier,
    lineColor: Color = MaterialTheme.colorScheme.outlineVariant,
    accentColor: Color = MaterialTheme.colorScheme.tertiary
) {
    Canvas(
        modifier = modifier
            .fillMaxWidth()
            .height(14.dp)
    ) {
        val cy = size.height / 2f
        val cx = size.width / 2f
        val jewelRadius = 4.5.dp.toPx()

        drawLine(
            color = lineColor,
            start = Offset(0f, cy),
            end = Offset(cx - jewelRadius * 2.4f, cy),
            strokeWidth = 1.dp.toPx()
        )
        drawLine(
            color = lineColor,
            start = Offset(cx + jewelRadius * 2.4f, cy),
            end = Offset(size.width, cy),
            strokeWidth = 1.dp.toPx()
        )

        val diamond = Path().apply {
            moveTo(cx, cy - jewelRadius)
            lineTo(cx + jewelRadius, cy)
            lineTo(cx, cy + jewelRadius)
            lineTo(cx - jewelRadius, cy)
            close()
        }
        drawPath(path = diamond, color = accentColor)
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable

    onToggleLessonCompleted: (String) -> Unit,
    onRecordQuizScore: (Int) -> Unit,
    onSpeakText: (String, String) -> Unit,
    modifier: Modifier = Modifier

    var selectedSection by remember { mutableStateOf(AcademySection.LESSONS) }
    var inspectedConcept by remember { mutableStateOf<StudyConcept?>(null) }

    Box(

    ) {
        LazyColumn(
            modifier = Modifier
                .fillMaxSize()
                .widthIn(max = 720.dp),
            contentPadding = PaddingValues(horizontal = 18.dp, vertical = 10.dp),
            verticalArrangement = Arrangement.spacedBy(16.dp)
        ) {
            // Illustrated Academy Banner
            item {
                Card(
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(164.dp),
                    shape = MaterialTheme.shapes.large
                ) {
                    Box(modifier = Modifier.fillMaxSize()) {
                        Image(

            item {

                    color = MaterialTheme.colorScheme.onSurface

                    color = MaterialTheme.colorScheme.primaryContainer.copy(alpha = 0.45f)

@Composable
private fun StudyLessonCard(
    lesson: StudyLesson,
    isCompleted: Boolean,
    onToggleComplete: () -> Unit
) {
    var expanded by remember { mutableStateOf(lesson.number == 1) }

    Surface(
        modifier = Modifier
            .fillMaxWidth()
            .testTag("study_lesson_${lesson.id}"),
        shape = MaterialTheme.shapes.large,
        color = MaterialTheme.colorScheme.surface,
        border = BorderStroke(
            width = if (isCompleted) 1.5.dp else 1.dp,
            color = if (isCompleted) MaterialTheme.colorScheme.primary else MaterialTheme.colorScheme.outlineVariant
        )
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .clickable { expanded = !expanded }
                .padding(18.dp),
            verticalArrangement = Arrangement.spacedBy(10.dp)
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.SpaceBetween
            ) {
                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    modifier = Modifier.weight(1f)
                ) {
                    RubElHizbBadge(numberText = "${lesson.number}", size = 40.dp)
                    Spacer(modifier = Modifier.width(12.dp))
                    Column {
                        Row(verticalAlignment = Alignment.CenterVertically) {
                            Text(
                                text = "${lesson.categoryTag.uppercase()} • ${lesson.level.uppercase()}",
                                style = MaterialTheme.typography.labelSmall,
                                color = MaterialTheme.colorScheme.tertiary,
                                fontWeight = FontWeight.Bold
                            )
                            if (isCompleted) {
                                Spacer(modifier = Modifier.width(6.dp))
                                Icon(
                                    imageVector = Icons.Rounded.CheckCircle,
                                    contentDescription = "Completed",
                                    tint = MaterialTheme.colorScheme.primary,
                                    modifier = Modifier.size(15.dp)
                                )
                            }
                        }
                        Text(
                            text = lesson.title,
                            style = MaterialTheme.typography.titleMedium,
                            color = MaterialTheme.colorScheme.onSurface
                        )
                    }
                }
                Icon(
                    imageVector = if (expanded) Icons.Rounded.ExpandLess else Icons.Rounded.ExpandMore,
                    contentDescription = null,
                    tint = MaterialTheme.colorScheme.onSurfaceVariant
                )
            }

            Text(
                text = lesson.summary,
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onSurfaceVariant
            )

            AnimatedVisibility(visible = expanded) {
                Column(verticalArrangement = Arrangement.spacedBy(12.dp)) {
                    OrnamentalDivider()
                    Text(
                        text = lesson.explanation,
                        style = MaterialTheme.typography.bodyMedium,
                        color = MaterialTheme.colorScheme.onSurface
                    )

                    Surface(
                        shape = MaterialTheme.shapes.medium,
                        color = MaterialTheme.colorScheme.surfaceContainerLow
                    ) {
                        Column(
                            modifier = Modifier.padding(14.dp),
                            verticalArrangement = Arrangement.spacedBy(6.dp)
                        ) {
                            Text(
                                text = "Key Takeaways",
                                style = MaterialTheme.typography.titleSmall,
                                color = MaterialTheme.colorScheme.primary
                            )
                            lesson.keyPoints.forEach { point ->
                                Text(
                                    text = "• $point",
                                    style = MaterialTheme.typography.bodySmall,
                                    color = MaterialTheme.colorScheme.onSurface
                                )
                            }
                        }
                    }

                    Text(
                        text = "Guided Examples",
                        style = MaterialTheme.typography.titleSmall,
                        color = MaterialTheme.colorScheme.primary
                    )

                    lesson.examples.forEach { ex ->
                        Surface(
                            modifier = Modifier.fillMaxWidth(),
                            shape = MaterialTheme.shapes.small,
                            color = MaterialTheme.colorScheme.primaryContainer.copy(alpha = 0.35f)
                        ) {
                            Column(
                                modifier = Modifier.padding(14.dp),
                                verticalArrangement = Arrangement.spacedBy(4.dp)
                            ) {
                                EditorialQuoteText(
                                    text = "“${ex.quote}”",
                                    fontSize = 16.sp,
                                    color = MaterialTheme.colorScheme.primary,
                                    modifier = Modifier.fillMaxWidth()
                                )
                                Text(
                                    text = ex.reference,
                                    style = MaterialTheme.typography.labelMedium,
                                    color = MaterialTheme.colorScheme.tertiary
                                )
                                Text(
                                    text = ex.takeaway,
                                    style = MaterialTheme.typography.bodySmall,
                                    color = MaterialTheme.colorScheme.onSurfaceVariant
                                )
                            }
                        }
                    }

                    FilledTonalButton(
                        onClick = onToggleComplete,
                        modifier = Modifier
                            .fillMaxWidth()
                            .testTag("complete_lesson_button_${lesson.id}")
                    ) {
                        Icon(
                            imageVector = Icons.Rounded.CheckCircle,
                            contentDescription = null,
                            modifier = Modifier.size(18.dp)
                        )
                        Spacer(modifier = Modifier.width(8.dp))
                        Text(if (isCompleted) "Lesson Completed (Tap to Undo)" else "Mark Lesson Complete")
                    }
                }
            }
        }
    }
}

@Composable
private fun AcademyQuizCard(
    bestScore: Int,
    onQuizFinished: (Int) -> Unit
) {
    val questions = AcademyData.quizQuestions
    var currentIndex by remember { mutableIntStateOf(0) }
    var selectedOption by remember { mutableStateOf<Int?>(null) }
    var score by remember { mutableIntStateOf(0) }
    var isFinished by remember { mutableStateOf(false) }

    if (isFinished) {
        Surface(
            modifier = Modifier
                .fillMaxWidth()
                .testTag("quiz_result_card"),
            shape = MaterialTheme.shapes.large,
            color = MaterialTheme.colorScheme.surface,
            border = BorderStroke(1.5.dp, MaterialTheme.colorScheme.tertiary)
        ) {
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(24.dp),
                horizontalAlignment = Alignment.CenterHorizontally,
                verticalArrangement = Arrangement.spacedBy(12.dp)
            ) {
                Icon(
                    imageVector = Icons.Rounded.EmojiEvents,
                    contentDescription = null,
                    tint = MaterialTheme.colorScheme.tertiary,
                    modifier = Modifier.size(52.dp)
                )
                Text(
                    text = "Well Done!",
                    style = MaterialTheme.typography.headlineMedium,
                    color = MaterialTheme.colorScheme.primary
                )
                Text(
                    text = "You scored $score out of ${questions.size}",
                    style = MaterialTheme.typography.titleLarge,
                    color = MaterialTheme.colorScheme.onSurface
                )
                Text(
                    text = "Personal Best: ${maxOf(bestScore, score)} / ${questions.size}",
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
                Spacer(modifier = Modifier.height(4.dp))
                Button(
                    onClick = {
                        currentIndex = 0
                        selectedOption = null
                        score = 0
                        isFinished = false
                    },
                    modifier = Modifier.testTag("quiz_restart_button")
                ) {
                    Icon(
                        imageVector = Icons.Rounded.Refresh,
                        contentDescription = null,
                        modifier = Modifier.size(18.dp)
                    )
                    Spacer(modifier = Modifier.width(8.dp))
                    Text("Retake Quiz")
                }
            }
        }
        return
    }

    val question = questions[currentIndex]
    val answered = selectedOption != null

    Surface(
        modifier = Modifier
            .fillMaxWidth()
            .testTag("quiz_active_card"),
        shape = MaterialTheme.shapes.large,
        color = MaterialTheme.colorScheme.surface,
        border = BorderStroke(1.dp, MaterialTheme.colorScheme.outlineVariant)
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(20.dp),
            verticalArrangement = Arrangement.spacedBy(14.dp)
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Surface(
                    shape = RoundedCornerShape(50),
                    color = MaterialTheme.colorScheme.tertiaryContainer
                ) {
                    Text(
                        text = question.topic,
                        style = MaterialTheme.typography.labelMedium,
                        color = MaterialTheme.colorScheme.onTertiaryContainer,
                        modifier = Modifier.padding(horizontal = 12.dp, vertical = 4.dp)
                    )
                }
                Text(
                    text = "Question ${currentIndex + 1} of ${questions.size} • Best: $bestScore",
                    style = MaterialTheme.typography.labelMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
            }

            LinearProgressIndicator(
                progress = { (currentIndex + 1).toFloat() / questions.size.toFloat() },
                modifier = Modifier
                    .fillMaxWidth()
                    .height(6.dp)
                    .clip(RoundedCornerShape(50))
            )

            question.featuredQuote?.let { quote ->
                EditorialQuoteText(
                    text = quote,
                    fontSize = 18.sp,
                    color = MaterialTheme.colorScheme.primary,
                    textAlign = TextAlign.Center,
                    italic = true,
                    modifier = Modifier.fillMaxWidth()
                )
            }

            Text(
                text = question.question,
                style = MaterialTheme.typography.titleMedium,
                color = MaterialTheme.colorScheme.onSurface
            )

            Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
                question.options.forEachIndexed { idx, option ->
                    val isSelected = selectedOption == idx
                    val isCorrect = idx == question.correctIndex

                    val containerColor = when {
                        !answered -> MaterialTheme.colorScheme.surfaceContainerLow
                        isCorrect -> MaterialTheme.colorScheme.primaryContainer
                        isSelected && !isCorrect -> MaterialTheme.colorScheme.errorContainer
                        else -> MaterialTheme.colorScheme.surfaceContainerLow.copy(alpha = 0.6f)
                    }
                    val borderColor = when {
                        !answered -> MaterialTheme.colorScheme.outlineVariant
                        isCorrect -> MaterialTheme.colorScheme.primary
                        isSelected && !isCorrect -> MaterialTheme.colorScheme.error
                        else -> MaterialTheme.colorScheme.outlineVariant
                    }

                    Surface(
                        modifier = Modifier
                            .fillMaxWidth()
                            .clip(MaterialTheme.shapes.small)
                            .clickable(enabled = !answered) {
                                selectedOption = idx
                                if (idx == question.correctIndex) {
                                    score += 1
                                }
                            }
                            .testTag("quiz_option_$idx"),
                        shape = MaterialTheme.shapes.small,
                        color = containerColor,
                        border = BorderStroke(1.2.dp, borderColor)
                    ) {
                        Text(
                            text = option,
                            style = MaterialTheme.typography.bodyLarge,
                            color = MaterialTheme.colorScheme.onSurface,
                            modifier = Modifier.padding(horizontal = 16.dp, vertical = 14.dp)
                        )
                    }
                }
            }

            AnimatedVisibility(visible = answered) {
                Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
                    Surface(
                        shape = MaterialTheme.shapes.small,
                        color = MaterialTheme.colorScheme.secondaryContainer.copy(alpha = 0.7f)
                    ) {
                        Text(
                            text = question.explanation,
                            style = MaterialTheme.typography.bodyMedium,
                            color = MaterialTheme.colorScheme.onSecondaryContainer,
                            modifier = Modifier.padding(14.dp)
                        )
                    }
                    Button(
                        onClick = {
                            if (currentIndex < questions.lastIndex) {
                                currentIndex += 1
                                selectedOption = null
                            } else {
                                onQuizFinished(score)
                                isFinished = true
                            }
                        },
                        modifier = Modifier
                            .fillMaxWidth()
                            .testTag("quiz_next_button")
                    ) {
                        Text(if (currentIndex < questions.lastIndex) "Next Question" else "Finish & See Score")
                    }
                }
            }

            if (!answered && currentIndex > 0) {
                OutlinedButton(
                    onClick = {
                        currentIndex = 0
                        selectedOption = null
                        score = 0
                    },
                    modifier = Modifier.align(Alignment.End)
                ) {
                    Text("Restart")
                }
            }
        }
    }
}

@Composable
fun DuaScreen(
    userProgress: UserProgress,
    duaSessionCounts: Map<String, Int>,
    speakingId: String?,
    onToggleBookmark: (String) -> Unit,
    onIncrementRepeat: (String, Int) -> Unit,
    onSpeakDua: (String, String) -> Unit,
    onOpenCounter: () -> Unit,
    modifier: Modifier = Modifier
) {
    val context = LocalContext.current
    var searchQuery by remember { mutableStateOf("") }
    var selectedCategory by remember { mutableStateOf(DuaCategory.ALL) }
    var showBookmarkedOnly by remember { mutableStateOf(false) }

    val filteredDuas = remember(searchQuery, selectedCategory, showBookmarkedOnly, userProgress.bookmarkedDuaIds) {
        HadithDuaData.duas.filter { dua ->
            val matchesQuery = searchQuery.isBlank() ||
                dua.title.contains(searchQuery, ignoreCase = true) ||
                dua.text.contains(searchQuery, ignoreCase = true) ||
                dua.occasion.contains(searchQuery, ignoreCase = true)
            val matchesCategory = selectedCategory == DuaCategory.ALL || dua.category == selectedCategory
            val matchesBookmark = !showBookmarkedOnly || dua.id in userProgress.bookmarkedDuaIds
            matchesQuery && matchesCategory && matchesBookmark
        }
    }

    Box(
        modifier = modifier.fillMaxSize(),
        contentAlignment = Alignment.TopCenter
    ) {
        LazyColumn(
            modifier = Modifier
                .fillMaxSize()
                .widthIn(max = 720.dp),
            contentPadding = PaddingValues(horizontal = 18.dp, vertical = 10.dp),
            verticalArrangement = Arrangement.spacedBy(14.dp)
        ) {
            item {
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(
                        text = "Heartfelt daily supplications organized by occasion with built-in recitation tracking.",
                        style = MaterialTheme.typography.bodyMedium,
                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                        modifier = Modifier.weight(1f)
                    )
                    Spacer(modifier = Modifier.width(10.dp))
                    FilledTonalButton(
                        onClick = onOpenCounter,
                        modifier = Modifier.testTag("dua_open_tasbih_button")
                    ) {
                        Icon(
                            imageVector = Icons.Rounded.TouchApp,
                            contentDescription = null,
                            modifier = Modifier.size(18.dp)
                        )
                        Spacer(modifier = Modifier.width(6.dp))
                        Text("Counter")
                    }
                }
            }

            item {
                OutlinedTextField(
                    value = searchQuery,
                    onValueChange = { searchQuery = it },
                    modifier = Modifier
                        .fillMaxWidth()
                        .testTag("dua_search_input"),
                    placeholder = { Text(stringResource(R.string.search_dua)) },
                    leadingIcon = { Icon(Icons.Rounded.Search, contentDescription = null) },
                    trailingIcon = {
                        if (searchQuery.isNotEmpty()) {
                            IconButton(onClick = { searchQuery = "" }) {
                                Icon(Icons.Rounded.Close, contentDescription = "Clear search")
                            }
                        }
                    },
                    singleLine = true,
                    shape = MaterialTheme.shapes.medium,
                    colors = OutlinedTextFieldDefaults.colors(
                        focusedContainerColor = MaterialTheme.colorScheme.surface,
                        unfocusedContainerColor = MaterialTheme.colorScheme.surface
                    )
                )
            }

            item {
                LazyRow(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                    item {
                        FilterChip(
                            selected = showBookmarkedOnly,
                            onClick = { showBookmarkedOnly = !showBookmarkedOnly },
                            label = { Text("Saved (${userProgress.bookmarkedDuaIds.size})") },
                            leadingIcon = {
                                Icon(
                                    imageVector = Icons.Rounded.Bookmark,
                                    contentDescription = null,
                                    modifier = Modifier.size(16.dp)
                                )
                            },
                            colors = FilterChipDefaults.filterChipColors(
                                selectedContainerColor = MaterialTheme.colorScheme.tertiary,
                                selectedLabelColor = MaterialTheme.colorScheme.onTertiary,
                                selectedLeadingIconColor = MaterialTheme.colorScheme.onTertiary
                            )
                        )
                    }
                    items(DuaCategory.entries) { category ->
                        FilterChip(
                            selected = selectedCategory == category,
                            onClick = { selectedCategory = category },
                            label = { Text(category.label) },
                            colors = FilterChipDefaults.filterChipColors(
                                selectedContainerColor = MaterialTheme.colorScheme.primary,
                                selectedLabelColor = MaterialTheme.colorScheme.onPrimary
                            )
                        )
                    }
                }
            }

            if (filteredDuas.isEmpty()) {
                item {
                    Surface(
                        modifier = Modifier.fillMaxWidth(),
                        shape = MaterialTheme.shapes.medium,
                        color = MaterialTheme.colorScheme.surfaceContainerLow
                    ) {
                        Column(
                            modifier = Modifier.padding(28.dp),
                            horizontalAlignment = Alignment.CenterHorizontally,
                            verticalArrangement = Arrangement.spacedBy(6.dp)
                        ) {
                            Text(
                                text = "No prayers match this filter",
                                style = MaterialTheme.typography.titleMedium
                            )
                            Text(
                                text = "Select another occasion category or bookmark your favorite prayers.",
                                style = MaterialTheme.typography.bodySmall,
                                color = MaterialTheme.colorScheme.onSurfaceVariant,
                                textAlign = TextAlign.Center
                            )
                        }
                    }
                }
            } else {
                items(filteredDuas, key = { it.id }) { dua ->
                    val isBookmarked = dua.id in userProgress.bookmarkedDuaIds
                    val currentCount = duaSessionCounts[dua.id] ?: 0
                    val speechKey = "dua_${dua.id}"

                    DuaCard(
                        dua = dua,
                        currentRepeatCount = currentCount,
                        isBookmarked = isBookmarked,
                        isSpeaking = speakingId == speechKey,
                        onIncrementRepeat = { onIncrementRepeat(dua.id, dua.recommendedCount) },
                        onToggleBookmark = { onToggleBookmark(dua.id) },
                        onListen = {
                            onSpeakDua(speechKey, "${dua.title}. ${dua.text}")
                        },
                        onShare = {
                            ShareUtils.shareText(
                                context,
                                dua.title,
                                "\"${dua.text}\"\n(${dua.occasion})\n— ${dua.reference}"
                            )
                        }
                    )
                }
            }

            item { Spacer(modifier = Modifier.height(16.dp)) }
        }
    }
}

@Composable
private fun DuaCard(
    dua: Dua,
    currentRepeatCount: Int,
    isBookmarked: Boolean,
    isSpeaking: Boolean,
    onIncrementRepeat: () -> Unit,
    onToggleBookmark: () -> Unit,
    onListen: () -> Unit,
    onShare: () -> Unit
) {
    val haptic = LocalHapticFeedback.current
    val isDone = currentRepeatCount >= dua.recommendedCount

    Surface(
        modifier = Modifier
            .fillMaxWidth()
            .testTag("dua_card_${dua.id}"),
        shape = MaterialTheme.shapes.large,
        color = MaterialTheme.colorScheme.surface,
        border = BorderStroke(
            width = if (isDone) 1.5.dp else 1.dp,
            color = if (isDone) MaterialTheme.colorScheme.primary else MaterialTheme.colorScheme.outlineVariant
        )
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(18.dp),
            verticalArrangement = Arrangement.spacedBy(10.dp)
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Surface(
                    shape = RoundedCornerShape(50),
                    color = MaterialTheme.colorScheme.secondaryContainer
                ) {
                    Text(
                        text = dua.category.label,
                        style = MaterialTheme.typography.labelMedium,
                        color = MaterialTheme.colorScheme.onSecondaryContainer,
                        modifier = Modifier.padding(horizontal = 12.dp, vertical = 4.dp)
                    )
                }

                Row(verticalAlignment = Alignment.CenterVertically) {
                    IconButton(onClick = onListen) {
                        Icon(
                            imageVector = if (isSpeaking) Icons.Rounded.Stop else Icons.AutoMirrored.Rounded.VolumeUp,
                            contentDescription = stringResource(
                                if (isSpeaking) R.string.cd_stop_audio else R.string.cd_listen
                            ),
                            tint = if (isSpeaking) MaterialTheme.colorScheme.tertiary else MaterialTheme.colorScheme.onSurfaceVariant
                        )
                    }
                    IconButton(onClick = onToggleBookmark) {
                        Icon(
                            imageVector = if (isBookmarked) Icons.Rounded.Bookmark else Icons.Outlined.BookmarkBorder,
                            contentDescription = stringResource(R.string.cd_bookmark),
                            tint = if (isBookmarked) MaterialTheme.colorScheme.tertiary else MaterialTheme.colorScheme.onSurfaceVariant
                        )
                    }
                    IconButton(onClick = onShare) {
                        Icon(
                            imageVector = Icons.Rounded.Share,
                            contentDescription = stringResource(R.string.cd_share),
                            tint = MaterialTheme.colorScheme.onSurfaceVariant
                        )
                    }
                }
            }

            Text(
                text = dua.title,
                style = MaterialTheme.typography.titleLarge,
                color = MaterialTheme.colorScheme.primary
            )

            Text(
                text = dua.occasion,
                style = MaterialTheme.typography.bodySmall.copy(fontStyle = FontStyle.Italic),
                color = MaterialTheme.colorScheme.tertiary
            )

            EditorialQuoteText(
                text = "“${dua.text}”",
                fontSize = 18.sp,
                color = MaterialTheme.colorScheme.onSurface,
                modifier = Modifier.fillMaxWidth()
            )

            OrnamentalDivider()

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = dua.reference,
                    style = MaterialTheme.typography.labelMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    modifier = Modifier.weight(1f)
                )

                FilledTonalButton(
                    onClick = {
                        haptic.performHapticFeedback(HapticFeedbackType.TextHandleMove)
                        onIncrementRepeat()
                    },
                    colors = ButtonDefaults.filledTonalButtonColors(
                        containerColor = if (isDone) {
                            MaterialTheme.colorScheme.primary
                        } else {
                            MaterialTheme.colorScheme.primaryContainer
                        },
                        contentColor = if (isDone) {
                            MaterialTheme.colorScheme.onPrimary
                        } else {
                            MaterialTheme.colorScheme.onPrimaryContainer
                        }
                    ),
                    modifier = Modifier.testTag("dua_repeat_button_${dua.id}")
                ) {
                    if (isDone) {
                        Icon(
                            imageVector = Icons.Rounded.CheckCircle,
                            contentDescription = null,
                            modifier = Modifier.size(16.dp)
                        )
                        Spacer(modifier = Modifier.width(6.dp))
                    }
                    Text(
                        text = if (isDone) {
                            "Completed (${dua.recommendedCount}/${dua.recommendedCount})"
                        } else {
                            "Recite $currentRepeatCount/${dua.recommendedCount}"
                        }
                    )
                }
            }
        }
    }
}

@Composable
private fun PillarCard(
    title: String,
    subtitle: String,
    icon: ImageVector,
    containerColor: Color,
    accentColor: Color,
    onClick: () -> Unit,
    modifier: Modifier = Modifier

                    .background(accentColor.copy(alpha = 0.14f)),
                contentAlignment = Alignment.Center
            ) {
                Icon(
                    imageVector = icon,
                    contentDescription = null,
                    tint = accentColor,
                    modifier = Modifier.size(22.dp)
                )
            }
            Column {
                Text(
                    text = title,
                    style = MaterialTheme.typography.titleMedium,
                    color = MaterialTheme.colorScheme.onSurface
                )
                Text(
                    text = subtitle,
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,

private enum class ChapterFilter(val label: String) {
    ALL("All Chapters"),
    COMPLETE("Complete"),
    STUDIED("Studied"),
    BOOKMARKED("Bookmarked")
}

@Composable
fun QuranListScreen(
    userProgress: UserProgress,
    onSelectSurah: (Int) -> Unit,
    modifier: Modifier = Modifier
) {
    var searchQuery by remember { mutableStateOf("") }
    var selectedFilter by remember { mutableStateOf(ChapterFilter.ALL) }

    val filteredSurahs = remember(searchQuery, selectedFilter, userProgress) {
        QuranData.surahs.filter { surah ->
            val matchesQuery = searchQuery.isBlank() ||
                surah.englishTitle.contains(searchQuery, ignoreCase = true) ||
                surah.classicName.contains(searchQuery, ignoreCase = true) ||
                surah.introduction.contains(searchQuery, ignoreCase = true) ||
                surah.number.toString() == searchQuery.trim()
            val matchesFilter = when (selectedFilter) {
                ChapterFilter.ALL -> true
                ChapterFilter.COMPLETE -> surah.isCompleteInApp
                ChapterFilter.STUDIED -> surah.number in userProgress.studiedSurahNumbers
                ChapterFilter.BOOKMARKED -> userProgress.bookmarkedAyahKeys.any {
                    it.startsWith("${surah.number}:")
                }
            }
            matchesQuery && matchesFilter
        }
    }

    Box(
        modifier = modifier.fillMaxSize(),
        contentAlignment = Alignment.TopCenter
    ) {
        LazyColumn(
            modifier = Modifier
                .fillMaxSize()
                .widthIn(max = 720.dp),
            contentPadding = PaddingValues(horizontal = 18.dp, vertical = 10.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp)

import com.example.data.HadithDuaData
import com.example.data.QuranData
import com.example.data.Surah
import com.example.data.UserProgress
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
import java.util.Calendar
import java.util.Locale

data class DailyInspiration(
    val surah: Surah,
    val ayah: Ayah,
    val hadith: Hadith,
    val dua: Dua
)

class AcademyViewModel(application: Application) : AndroidViewModel(application) {

    private val repository = AcademyPreferencesRepository(application.applicationContext)

    val userProgress: StateFlow<UserProgress> = repository.userProgressFlow.stateIn(
        scope = viewModelScope,
        started = SharingStarted.WhileSubscribed(5_000),
        initialValue = UserProgress()
    )

    private val _speakingId = MutableStateFlow<String?>(null)
    val speakingId: StateFlow<String?> = _speakingId.asStateFlow()

    private val _duaSessionCounts = MutableStateFlow<Map<String, Int>>(emptyMap())
    val duaSessionCounts: StateFlow<Map<String, Int>> = _duaSessionCounts.asStateFlow()

    private var tts: TextToSpeech? = null
    private var ttsReady = false

    val dailyInspiration: DailyInspiration by lazy {
        val dayOfYear = Calendar.getInstance().get(Calendar.DAY_OF_YEAR)
        val allVerses = QuranData.surahs.flatMap { surah ->
            surah.ayahs.map { ayah -> surah to ayah }
        }
        val (dailySurah, dailyAyah) = allVerses[dayOfYear % allVerses.size]
        val dailyHadith = HadithDuaData.hadiths[dayOfYear % HadithDuaData.hadiths.size]
        val dailyDua = HadithDuaData.duas[dayOfYear % HadithDuaData.duas.size]
        DailyInspiration(dailySurah, dailyAyah, dailyHadith, dailyDua)
    }

    init {
        initTts()
    }

    private fun initTts() {
        tts = TextToSpeech(getApplication()) { status ->
            if (status == TextToSpeech.SUCCESS) {
                ttsReady = true
                tts?.language = Locale.US
                tts?.setOnUtteranceProgressListener(object : UtteranceProgressListener() {
                    override fun onStart(utteranceId: String?) {}
                    override fun onDone(utteranceId: String?) {
                        _speakingId.value = null
                    }
                    @Deprecated("Deprecated in Java")
                    override fun onError(utteranceId: String?) {
                        _speakingId.value = null
                    }
                })
            }
        }
    }

    fun speakText(id: String, text: String) {
        if (!ttsReady) return
        if (_speakingId.value == id) {
            stopSpeaking()
            return
        }
        val engine = tts ?: return
        engine.language = Locale.US
        engine.setSpeechRate(0.92f)
        _speakingId.value = id
        engine.speak(text, TextToSpeech.QUEUE_FLUSH, null, id)
    }

    fun stopSpeaking() {
        tts?.stop()
        _speakingId.value = null
    }

    fun openSurah(surahNumber: Int) {
        viewModelScope.launch { repository.setLastReadSurah(surahNumber) }
    }

    fun toggleAyahBookmark(surahNumber: Int, ayahNumber: Int) {
        viewModelScope.launch { repository.toggleAyahBookmark("$surahNumber:$ayahNumber") }
    }

    fun toggleHadithBookmark(hadithId: String) {
        viewModelScope.launch { repository.toggleHadithBookmark(hadithId) }
    }

    fun toggleDuaBookmark(duaId: String) {
        viewModelScope.launch { repository.toggleDuaBookmark(duaId) }
    }

    fun toggleSurahStudied(surahNumber: Int) {
        viewModelScope.launch { repository.toggleSurahStudied(surahNumber) }
    }

    fun toggleLessonCompleted(lessonId: String) {
        viewModelScope.launch { repository.toggleLessonCompleted(lessonId) }
    }

    fun incrementCounter() {
        viewModelScope.launch { repository.incrementCounter() }
    }

    fun resetCounter() {
        viewModelScope.launch { repository.resetCounter() }
    }

    fun setCounterTarget(target: Int) {
        viewModelScope.launch { repository.setCounterTarget(target) }
    }

    fun incrementDuaRepeat(duaId: String, recommendedCount: Int) {
        _duaSessionCounts.update { map ->
            val current = map[duaId] ?: 0
            val next = if (current >= recommendedCount) 0 else current + 1
            map + (duaId to next)
        }
    }

    fun recordQuizScore(score: Int) {
        viewModelScope.launch { repository.recordQuizScore(score) }
    }

    fun adjustReadingFontSize(deltaSp: Float) {
        val current = userProgress.value.readingFontSizeSp
        viewModelScope.launch { repository.setReadingFontSize(current + deltaSp) }
    }

    fun toggleStudyNotes() {
        val current = userProgress.value.showStudyNotes
        viewModelScope.launch { repository.setShowStudyNotes(!current) }
    }

    fun toggleDarkMode() {
        viewModelScope.launch { repository.toggleDarkMode() }
    }

    override fun onCleared() {
        tts?.stop()
        tts?.shutdown()
        super.onCleared()
    }
}
